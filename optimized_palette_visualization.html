Elixir Code Example

# Generate an optimized 10-color palette
colors = ChromaEx.generate_palette(10,
  target_colors: [
    "#4269d0", "#efb118", "#ff725c", "#6cc5b0", "#3ca951"
  ],
  weights: %{
    energy: 1.25,
    range: 1.0,
    target: 0.75,
    deuteranomaly: 0.5
  }
)

# Convert to hex codes for use in your application
hex_colors = Enum.map(colors, &ChromaEx.to_hex/1)
